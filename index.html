require('dotenv').config(); const { Client, GatewayIntentBits, EmbedBuilder, PermissionsBitField } = require('discord.js'); const express = require('express'); const app = express(); app.use(express.json()); const client = new Client({ intents: [ GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages, GatewayIntentBits.MessageContent ] }); // --- COMMAND LISTENER (,c command) --- client.on('messageCreate', async (message) => { // Ignore bot messages or messages without prefix if (message.author.bot || !message.content.startsWith(',c')) return; // Check if user has permission to manage messages if (!message.member.permissions.has(PermissionsBitField.Flags.ManageMessages)) { return message.reply("You don't have permission to manage messages!"); } const args = message.content.trim().split(/ +/); let amount = parseInt(args[1]); // If no number is given, default to clearing 100 messages (Discord bulkDelete max per request) if (isNaN(amount)) { amount = 100; } else { // Account for the user's command message itself amount = amount + 1; } if (amount < 1 || amount > 100) { return message.reply('Please specify a number between 1 and 99.'); } try { const deleted = await message.channel.bulkDelete(amount, true); const count = deleted.size - 1 > 0 ? deleted.size - 1 : deleted.size; const reply = await message.channel.send(`🧹 Successfully cleared **${count}** messages!`); // Auto-delete the bot's confirmation message after 3 seconds setTimeout(() => reply.delete().catch(() => {}), 3000); } catch (error) { console.error('Failed to clear messages:', error); message.reply('There was an error trying to delete messages in this channel. (Note: Discord cannot bulk delete messages older than 14 days).'); } }); // --- GITHUB WEBHOOK LISTENER --- app.post('/github-webhook', async (req, res) => { const event = req.headers['x-github-event']; const payload = req.body; try { const channel = await client.channels.fetch(process.env.DISCORD_CHANNEL_ID); if (!channel) return res.status(404).send('Channel not found'); if (event === 'push') { const pusher = payload.pusher.name; const repo = payload.repository.name; const branch = payload.ref.replace('refs/heads/', ''); const commits = payload.commits || []; for (const commit of commits) { const htmlFiles = [ ...commit.added, ...commit.modified, ...commit.removed ].filter(file => file.endsWith('.html')); if (htmlFiles.length > 0) { const embed = new EmbedBuilder() .setColor(0xE34F26) .setTitle(`📄 HTML Update in ${repo}`) .setURL(commit.url) .setAuthor({ name: pusher, iconURL: payload.sender.avatar_url }) .setDescription(`**Commit message:** ${commit.message}\n**Branch:** \`${branch}\``) .addFields( { name: 'Changed HTML Files', value: htmlFiles.map(f => `\`${f}\``).join('\n') } ) .setTimestamp(new Date(commit.timestamp)); await channel.send({ embeds: [embed] }); } } } res.status(200).send('Webhook processed'); } catch (error) { console.error('Error delivering message:', error); res.status(500).send('Server Error'); } }); client.once('ready', () => { console.log(`Bot logged in as ${client.user.tag}`); app.listen(process.env.PORT || 3000, () => { console.log(`Webhook server running on port ${process.env.PORT || 3000}`); }); }); client.login(process.env.DISCORD_BOT_TOKEN);
